# LoomQ runtime dependencies. Every version is pinned exactly, as the rules
# require. Python 3.10, matching the official base image (spinqit only ships
# cp310 wheels).
#
# The three vendor SDKs do not coexist at their latest versions. spinqit and
# amazon-braket-default-simulator both hard-pin antlr4-python3-runtime, to
# incompatible releases, and each one's antlr-generated parser fails at
# *import* time under the other's runtime:
#
#   spinqit 0.2.4                          -> antlr4-python3-runtime==4.9.2
#   amazon-braket-default-simulator >=1.28 -> antlr4-python3-runtime==4.13.2
#
# 1.27.0 is the last Braket simulator on the 4.9.2 line, and 1.97.0 is the SDK
# released alongside it. That combination is what makes all three backends
# importable in one interpreter.

amazon-braket-sdk==1.97.0
amazon-braket-default-simulator==1.27.0
antlr4-python3-runtime==4.9.2

spinqit==0.2.4

pyqpanda==3.8.5

# Image size note. spinqit pulls in torch, and on Linux pip resolves torch to
# the CUDA build: measured on Ubuntu with python:3.10 the install is 5.2 GB, of
# which 3.4 GB is NVIDIA CUDA libraries that nothing in this project uses.
# Switching to the CPU-only wheel brings it to 1.4 GB and installs faster
# (66s vs 84s on the same machine), with all three SDKs still importing.
#
# It is left off by default because it adds a second package host that has to
# be reachable at build time, and a build failure costs more than a large
# image. To enable it, uncomment these two lines:
#
# --extra-index-url https://download.pytorch.org/whl/cpu
# torch==2.13.0+cpu
